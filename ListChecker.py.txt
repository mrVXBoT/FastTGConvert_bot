def new_session() -> dict:
    return {
        "file1": None, "file1_name": "",
        "file2": None, "file2_name": "",
        "step": 0,
        "active": False,
    }


def cleanup_temp(chat_id: int):
    folder = os.path.join(TEMP_ROOT, str(chat_id))
    if os.path.exists(folder):
        shutil.rmtree(folder, ignore_errors=True)
        logger.info("Cleaned temp folder for chat %s", chat_id)


def is_tdata_folder(entry: str) -> bool:
    """
    Tdata ফোল্ডার চেনার নিয়ম:
      - entry শেষে / বা \\ থাকবে (মানে এটা ফোল্ডার)
      - ফোল্ডারের নামে কোনো extension (.xxx) নেই
      উদাহরণ: "596696702021/" → True
               "596696702021/files/" → False (nested)
               "abc.zip/" → False (extension আছে)
    """
    # শুধু top-level ফোল্ডার (একটাই / থাকবে, একদম শেষে)
    normalized = entry.replace("\\", "/")
    if not normalized.endswith("/"):
        return False
    parts = normalized.rstrip("/").split("/")
    if len(parts) != 1:
        return False  # nested ফোল্ডার বাদ
    folder_name = parts[0]
    # extension নেই মানে Tdata
    return "." not in folder_name


def get_entry_top_folder(entry: str) -> str | None:
    """
    একটি ফাইল entry থেকে top-level ফোল্ডারের নাম বের করে।
    যেমন: "596696702021/D877F783" → "596696702021"
           "abc.session" → None (কোনো ফোল্ডার নেই)
    """
    normalized = entry.replace("\\", "/")
    parts = normalized.split("/")
    if len(parts) >= 2:
        return parts[0]
    return None


def classify_entry(entry: str) -> str | None:
    """
    আর্কাইভের একটি entry কে classify করে।
    Returns: 'tdata' | 'session' | 'json' | None
    None মানে এই entry বাদ দেওয়া হবে।
    """
    lower = entry.replace("\\", "/").lower()

    # .session ফাইল (যেকোনো জায়গায় থাকুক)
    if lower.endswith(".session"):
        return "session"

    # .json ফাইল
    if lower.endswith(".json"):
        return "json"

    # Tdata top-level ফোল্ডার
    if is_tdata_folder(entry):
        return "tdata"

    return None


def list_archive_all(data: bytes, filename: str) -> list[str]:
    """আর্কাইভের সব entry (ফাইল + ফোল্ডার) লিস্ট করে।"""
    buf = io.BytesIO(data)
    ext = os.path.splitext(filename)[1].lower()

    if ext == ".zip":
        with zipfile.ZipFile(buf) as zf:
            return zf.namelist()
    elif ext == ".rar":
        if not RAR_SUPPORTED:
            raise RuntimeError("rarfile library is not installed.")
        with rarfile.RarFile(buf) as rf:
            return rf.namelist()
    else:
        raise ValueError(f"Unsupported format: {ext}")


def extract_all_files(data: bytes, filename: str) -> dict[str, bytes]:
    """আর্কাইভের সব ফাইলের bytes বের করে dict-এ রাখে।"""
    buf = io.BytesIO(data)
    ext = os.path.splitext(filename)[1].lower()
    result = {}

    if ext == ".zip":
        with zipfile.ZipFile(buf) as zf:
            for name in zf.namelist():
                if not name.endswith("/"):
                    result[name] = zf.read(name)
    elif ext == ".rar":
        with rarfile.RarFile(buf) as rf:
            for name in rf.namelist():
                if not name.endswith("/") and not name.endswith("\\"):
                    result[name] = rf.read(name)
    return result


def build_result_zip(
    data1: bytes, name1: str,
    data2: bytes, name2: str,
) -> tuple[bytes, int, int, int, int]:
    """
    দুই আর্কাইভ তুলনা করে কমন item দিয়ে matched_files.zip বানায়।

    তিনটা ক্যাটাগরি:
      • Tdata  → numeric/নামের ফোল্ডার (extension নেই)
      • .session ফাইল
      • .json ফাইল

    Returns:
        (zip_bytes, tdata_count, session_count, json_count, total)
    """
    entries1 = list_archive_all(data1, name1)
    entries2 = list_archive_all(data2, name2)

    # ──────────────────────────────────────────────
    # Archive 1 থেকে maps তৈরি
    # ──────────────────────────────────────────────

    # Tdata: folder_name → [list of file entries inside]
    tdata_map1: dict[str, list[str]] = {}

    # Session: base_name_no_ext → entry_path
    session_map1: dict[str, str] = {}

    # JSON: base_name_no_ext → entry_path
    json_map1: dict[str, str] = {}

    for entry in entries1:
        kind = classify_entry(entry)
        if kind == "tdata":
            folder_name = entry.replace("\\", "/").rstrip("/").split("/")[-1]
            if folder_name not in tdata_map1:
                tdata_map1[folder_name] = []

        elif kind == "session":
            base = os.path.basename(entry.replace("\\", "/"))
            key = os.path.splitext(base)[0]
            if key not in session_map1:
                session_map1[key] = entry

        elif kind == "json":
            base = os.path.basename(entry.replace("\\", "/"))
            key = os.path.splitext(base)[0]
            if key not in json_map1:
                json_map1[key] = entry

    # Tdata ফোল্ডারের ভেতরের সব ফাইল খুঁজে বের করো
    for entry in entries1:
        normalized = entry.replace("\\", "/")
        if normalized.endswith("/"):
            continue  # ফোল্ডার entry বাদ
        top = get_entry_top_folder(normalized)
        if top and top in tdata_map1:
            tdata_map1[top].append(entry)

    # ──────────────────────────────────────────────
    # Archive 2 থেকে key sets তৈরি
    # ──────────────────────────────────────────────
    tdata_keys2: set[str] = set()
    session_keys2: set[str] = set()
    json_keys2: set[str] = set()

    for entry in entries2:
        kind = classify_entry(entry)
        if kind == "tdata":
            folder_name = entry.replace("\\", "/").rstrip("/").split("/")[-1]
            tdata_keys2.add(folder_name)
        elif kind == "session":
            base = os.path.basename(entry.replace("\\", "/"))
            session_keys2.add(os.path.splitext(base)[0])
        elif kind == "json":
            base = os.path.basename(entry.replace("\\", "/"))
            json_keys2.add(os.path.splitext(base)[0])

    # ──────────────────────────────────────────────
    # Common items
    # ──────────────────────────────────────────────
    common_tdata = set(tdata_map1.keys()) & tdata_keys2
    common_sessions = set(session_map1.keys()) & session_keys2
    common_jsons = set(json_map1.keys()) & json_keys2

    tdata_count = len(common_tdata)
    session_count = len(common_sessions)
    json_count = len(common_jsons)
    total = tdata_count + session_count + json_count

    # ──────────────────────────────────────────────
    # Result ZIP বানাও
    # ──────────────────────────────────────────────
    all_files1 = extract_all_files(data1, name1)
    out_buf = io.BytesIO()

    with zipfile.ZipFile(out_buf, "w", zipfile.ZIP_DEFLATED) as out_zip:

        # Tdata ফোল্ডার: পুরো ফোল্ডার সহ (স্ট্রাকচার রেখে)
        for folder_name in sorted(common_tdata):
            for file_entry in tdata_map1[folder_name]:
                if file_entry in all_files1:
                    # ZIP-এ ফোল্ডার স্ট্রাকচার বজায় রাখো
                    arc_name = file_entry.replace("\\", "/")
                    out_zip.writestr(arc_name, all_files1[file_entry])

        # .session ফাইল
        for key in sorted(common_sessions):
            entry_path = session_map1[key]
            if entry_path in all_files1:
                out_zip.writestr(
                    os.path.basename(entry_path.replace("\\", "/")),
                    all_files1[entry_path]
                )

        # .json ফাইল
        for key in sorted(common_jsons):
            entry_path = json_map1[key]
            if entry_path in all_files1:
                out_zip.writestr(
                    os.path.basename(entry_path.replace("\\", "/")),
                    all_files1[entry_path]
                )

    return out_buf.getvalue(), tdata_count, session_count, json_count, total

@bot.message_handler(content_types=["document"])
def handle_document(message):
    chat_id = message.chat.id
    sess = sessions.get(chat_id)

    if not sess or not sess.get("active"):
        bot.send_message(
            chat_id,
            "⚠️ আগে <b>File Matcher</b> সিলেক্ট করো।\n"
            "/start দিয়ে শুরু করো।",
        )
        return

    doc = message.document
    filename = doc.file_name or "file"
    ext = os.path.splitext(filename)[1].lower()

    if ext not in (".zip", ".rar"):
        bot.send_message(
            chat_id,
            "❌ শুধুমাত্র <b>.zip</b> বা <b>.rar</b> ফাইল সাপোর্টেড।",
            reply_markup=cancel_keyboard(),
        )
        return

    if ext == ".rar" and not RAR_SUPPORTED:
        bot.send_message(
            chat_id,
            "❌ <b>RAR সাপোর্ট পাওয়া যাচ্ছে না।</b>\n"
            "Server-এ <code>pip install rarfile</code> এবং "
            "<code>unrar</code> ইনস্টল করতে হবে।",
        )
        return

    # ─── ফাইল ডাউনলোড ───
    wait_msg = bot.send_message(chat_id, "⏳ ফাইল ডাউনলোড হচ্ছে...")

    try:
        file_info = bot.get_file(doc.file_id)
        downloaded = bot.download_file(file_info.file_path)
    except Exception as e:
        logger.exception("Download error")
        bot.edit_message_text(
            f"❌ ডাউনলোড ব্যর্থ হয়েছে:\n<code>{e}</code>",
            chat_id, wait_msg.message_id
        )
        return

    bot.delete_message(chat_id, wait_msg.message_id)

    # ─── Step 1: প্রথম ফাইল ───
    if sess["step"] == 1:
        sess["file1"] = downloaded
        sess["file1_name"] = filename
        sess["step"] = 2

        bot.send_message(
            chat_id,
            f"✅ <b>File 1 received:</b> <code>{filename}</code>\n\n"
            "Step 2️⃣ — এখন দ্বিতীয় ZIP বা RAR ফাইলটি পাঠাও:",
            reply_markup=cancel_keyboard(),
        )
        return

    # ─── Step 2: দ্বিতীয় ফাইল ───
    if sess["step"] == 2:
        sess["file2"] = downloaded
        sess["file2_name"] = filename
        sess["step"] = "processing"

        proc_msg = bot.send_message(chat_id, "⚙️ <b>Processing... একটু অপেক্ষা করো।</b>")

        try:
            zip_bytes, tdata, session_cnt, json_cnt, total = build_result_zip(
                sess["file1"], sess["file1_name"],
                sess["file2"], sess["file2_name"],
            )
        except Exception as e:
            logger.exception("Processing error")
            bot.edit_message_text(
                f"❌ প্রসেসিং ব্যর্থ হয়েছে:\n<code>{e}</code>",
                chat_id, proc_msg.message_id,
            )
            sessions[chat_id] = new_session()
            return

        bot.edit_message_text(
            f"🔍 <b>Found {total} matches</b> — ZIP তৈরি হচ্ছে...",
            chat_id, proc_msg.message_id,
        )

        if total == 0:
            bot.send_message(
                chat_id,
                "😕 <b>কোনো কমন item পাওয়া যায়নি।</b>\n\n"
                "দুই আর্কাইভে একই নামের কোনো Tdata ফোল্ডার, "
                ".session বা .json ফাইল নেই।",
                reply_markup=main_menu_keyboard(),
            )
        else:
            result_caption = (
                "✅ <b>Matched Result</b>\n\n"
                f"📁 Tdata folders  : <b>{tdata}</b>\n"
                f"🔑 Session files  : <b>{session_cnt}</b>\n"
                f"📄 Json files     : <b>{json_cnt}</b>\n"
                f"━━━━━━━━━━━━━━━━━━\n"
                f"✅ Total          : <b>{total} common items</b>"
            )
            bot.send_document(
                chat_id,
                document=io.BytesIO(zip_bytes),
                visible_file_name="matched_files.zip",
                caption=result_caption,
            )

        cleanup_temp(chat_id)
        sessions[chat_id] = new_session()

        bot.send_message(
            chat_id,
            "আবার ব্যবহার করতে নিচের মেনু থেকে সিলেক্ট করো 👇",
            reply_markup=main_menu_keyboard(),
        )
        return

    # অপ্রত্যাশিত অবস্থা
    bot.send_message(chat_id, "⚠️ অপ্রত্যাশিত অবস্থা। /start দিয়ে আবার চেষ্টা করো।")
    sessions[chat_id] = new_session()
