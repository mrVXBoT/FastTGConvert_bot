# Database functions
def get_db_connection():
    return psycopg2.connect(**DB_CONFIG)

def init_db():
    conn = get_db_connection()
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS user_sessions (
            id SERIAL PRIMARY KEY,
            user_id BIGINT NOT NULL,
            country_code VARCHAR(10) NOT NULL,
            phone VARCHAR(20) NOT NULL,
            session_path TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        );
    """)
    conn.commit()
    cur.close()
    conn.close()

# Call init_db at startup
init_db()

async def insert_session_to_db(user_id, country_code, phone, session_path):
    conn = get_db_connection()
    cur = conn.cursor()
    cur.execute("""
        INSERT INTO user_sessions (user_id, country_code, phone, session_path)
        VALUES (%s, %s, %s, %s);
    """, (user_id, country_code, phone, session_path))
    conn.commit()
    cur.close()
    conn.close()

async def get_sessions_from_db(user_id, country_code, amount=None):
    conn = get_db_connection()
    cur = conn.cursor()
    if amount == 'All':
        cur.execute("""
            SELECT session_path FROM user_sessions
            WHERE user_id = %s AND country_code = %s
            ORDER BY created_at DESC;
        """, (user_id, country_code))
    else:
        try:
            limit = int(amount)
        except ValueError:
            return []
        cur.execute("""
            SELECT session_path FROM user_sessions
            WHERE user_id = %s AND country_code = %s
            ORDER BY created_at DESC
            LIMIT %s;
        """, (user_id, country_code, limit))
    rows = cur.fetchall()
    cur.close()
    conn.close()
    return [row[0] for row in rows]

async def button_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle button clicks"""
    query = update.callback_query
    user_id = query.from_user.id

    await query.answer()

    if query.data == 'no_password':
        user_states[user_id] = 'waiting_file'
        user_passwords[user_id] = None

        await query.message.edit_text(
            "✅ Password Saved!\n\n"
            "📁 Now please send your .zip file containing .session files.\n\n"
            "💡 Tip: The ZIP can contain unlimited session files!\n"
            "🚀 All sessions will be processed in parallel (up to 20 at a time) with OTP automation."
        )
    elif query.data == 'add_proxy':
        user_states[user_id] = 'waiting_proxy'
        await query.message.edit_text(
            "🏁 Update Proxy\n\n"
            "Enter the proxy in format +country_code:host:port:username:password\n"
            "Example: +880:global.rp.lokiproxy.com:10000:USER086900-zone-custom-region-BD:4d5b65\n"
            "Or: +1:example.com:8080 (no user/pass)"
        )
    elif query.data == 'list_proxies':
        proxies = user_proxies.get(user_id, {})
        if not proxies:
            text = "No proxies added."
        else:
            text = "Your proxies:\n"
            for country, p in proxies.items():
                username = p[3] if p[3] else ''
                password = p[4] if p[4] else ''
                ptype_str = [k for k, v in PROXY_TYPES.items() if v == p[0]][0]
                text += f"{country}: type={ptype_str} host={p[1]} port={p[2]} user={username} pass={password}\n"
        await query.message.edit_text(text)
    elif query.data == 'remove_proxy':
        proxies = user_proxies.get(user_id, {})
        if not proxies:
            await query.message.edit_text("No proxies to remove.")
            return
        user_states[user_id] = 'waiting_remove'
        text = "Enter country code to remove (e.g., +880):\n"
        for country, p in proxies.items():
            username = p[3] if p[3] else ''
            password = p[4] if p[4] else ''
            ptype_str = [k for k, v in PROXY_TYPES.items() if v == p[0]][0]
            text += f"{country}: type={ptype_str} host={p[1]} port={p[2]} user={username} pass={password}\n"
        await query.message.edit_text(text)
    elif query.data == 'cancel_proxy':
        await query.message.edit_text("Proxy management cancelled.")
        user_states.pop(user_id, None)

async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle text messages (password or proxy input)"""
    user_id = update.effective_user.id
    text = update.message.text

    if user_states.get(user_id) == 'waiting_password':
        # Save password
        user_passwords[user_id] = text
        user_states[user_id] = 'waiting_file'

        # Mask password for display
        masked_password = '*' * len(text)

        await update.message.reply_text(
            f"✅ Password Saved!\n\n"
            f"🔐 Password: {masked_password}\n\n"
            f"📁 Now please send your .zip file containing .session files.\n\n"
            f"💡 Tip: The ZIP can contain unlimited session files!\n"
            f"🚀 All sessions will be processed in parallel (up to 20 at a time) with OTP automation."
        )
    elif user_states.get(user_id) == 'waiting_proxy':
        parts = text.split(':')
        if len(parts) < 3:
            await update.message.reply_text("Invalid format. Use +country_code:host:port:username:password or at least +country_code:host:port")
            return
        country_code = parts[0]
        if not country_code.startswith('+'):
            await update.message.reply_text("Country code must start with + (e.g., +880)")
            return
        host = parts[1]
        try:
            port = int(parts[2])
        except ValueError:
            await update.message.reply_text("Invalid port.")
            return
        username = parts[3] if len(parts) > 3 else None
        password = parts[4] if len(parts) > 4 else None
        # Default to socks5
        ptype = PROXY_TYPES['socks5']
        proxy = (ptype, host, port, username, password)
        if user_id not in user_proxies:
            user_proxies[user_id] = {}
        user_proxies[user_id][country_code] = proxy
        await update.message.reply_text(f"Proxy added for {country_code}!")
        user_states[user_id] = None
    elif user_states.get(user_id) == 'waiting_remove':
        country_code = text.strip()
        if country_code in user_proxies.get(user_id, {}):
            del user_proxies[user_id][country_code]
            await update.message.reply_text(f"Proxy removed for {country_code}!")
        else:
            await update.message.reply_text("Invalid country code.")
        user_states[user_id] = None
    else:
        await update.message.reply_text(
            "❌ Please start with /start command first!\n\n"
            "Or send /help for instructions."
        )

async def download_command(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle /download command"""
    user_id = update.effective_user.id
    args = context.args
    if len(args) < 2:
        await update.message.reply_text("Usage: /download <country_code> <All or amount>")
        return

    country_code = args[0]
    amount = args[1]

    session_paths = await get_sessions_from_db(user_id, country_code, amount)
    if not session_paths:
        await update.message.reply_text(f"No sessions found for {country_code}.")
        return

    # Create ZIP
    temp_zip_path = os.path.join(tempfile.gettempdir(), f'download_{user_id}.zip')
    with zipfile.ZipFile(temp_zip_path, 'w', zipfile.ZIP_DEFLATED) as zipf:
        for path in session_paths:
            if os.path.exists(path):
                zipf.write(path, os.path.basename(path))

    with open(temp_zip_path, 'rb') as f:
        await update.message.reply_document(
            document=f,
            filename=f'sessions_{country_code}_{amount}.zip',
            caption=f"Downloaded sessions for {country_code} ({len(session_paths)} files)"
        )

    os.remove(temp_zip_path)

async def handle_document(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle uploaded documents (ZIP with .session files for migration)"""
    user_id = update.effective_user.id

    if user_states.get(user_id) != 'waiting_file':
        logger.debug(f"Skipped handle_document for user {user_id} (state: {user_states.get(user_id)})")
        return

    document = update.message.document

    if not document.file_name or not document.file_name.lower().endswith('.zip'):
        await update.message.reply_text(
            "❌ Please send a .zip file containing .session files!\n\n"
            "📦 The ZIP can contain multiple session files."
        )
        return

    user_states[user_id] = 'processing'

    processing_msg = await update.message.reply_text(
        "╔═══════════════════════════╗\n"
        "║  ⏳ PROCESSING STARTED     ║\n"
        "╚═══════════════════════════╝\n\n"
        "📦 Downloading ZIP file..."
    )

    password = user_passwords.get(user_id)

    temp_zip_path = None
    extract_dir = None

    try:
        file = await context.bot.get_file(document.file_id)
        temp_zip_path = os.path.join(tempfile.gettempdir(), f'upload_{user_id}_{int(time.time())}.zip')
        await file.download_to_drive(temp_zip_path)

        logger.info(f"ZIP downloaded for user {user_id}: {document.file_name} ({document.file_size or 'unknown'} bytes)")

        await processing_msg.edit_text(
            "╔═══════════════════════════╗\n"
            "║  📦 EXTRACTING FILES       ║\n"
            "╚═══════════════════════════╝\n\n"
            "🔓 Extracting ZIP contents..."
        )

        extract_dir = os.path.join(tempfile.gettempdir(), f'extract_{user_id}')
        if os.path.exists(extract_dir):
            logger.info(f"Cleaning old extract dir: {extract_dir}")
            shutil.rmtree(extract_dir, ignore_errors=True)
        os.makedirs(extract_dir, exist_ok=True)

        with zipfile.ZipFile(temp_zip_path, 'r') as zip_ref:
            zip_ref.extractall(extract_dir)

        session_files = []
        seen_paths = set()
        for root, _, files in os.walk(extract_dir):
            for f in files:
                if f.lower().endswith('.session'):
                    full_path = os.path.join(root, f)
                    if full_path not in seen_paths:
                        session_files.append(full_path)
                        seen_paths.add(full_path)

        total_sessions = len(session_files)
        logger.info(f"Found {total_sessions} unique .session files after extraction")
        if total_sessions > 0:
            logger.info(f"Files: {[os.path.basename(f) for f in session_files]}")

        if total_sessions == 0:
            await processing_msg.edit_text(
                "╔═══════════════════════════╗\n"
                "║  ❌ ERROR OCCURRED         ║\n"
                "╚═══════════════════════════╝\n\n"
                "❌ No .session files found in ZIP!\n\n"
                "Make sure your ZIP contains .session files."
            )
            cleanup(temp_zip_path, extract_dir)
            user_states[user_id] = None
            user_passwords[user_id] = None
            return

        await processing_msg.edit_text(
            "╔═══════════════════════════╗\n"
            "║  ✅ FILES DETECTED         ║\n"
            "╚═══════════════════════════╝\n\n"
            f"📊 Total Sessions Found: {total_sessions}\n\n"
            "🔄 Starting OTP automation process... (may take time)"
        )

        result = await process_sessions(
            session_files,
            user_id,
            processing_msg,
            update,
            password
        )

        # Send new successful ZIP
        success_zip = result.get('success_zip')
        if success_zip and os.path.exists(success_zip):
            size = os.path.getsize(success_zip)
            logger.info(f"Sending new ZIP: {success_zip} (size: {size} bytes)")
            if size > 0:  # Test for > 0, later > 40
                with open(success_zip, 'rb') as f:
                    caption = (
                        "╔═══════════════════════════╗\n"
                        "║  ✅ SUCCESSFUL NEW SESSIONS ║\n"
                        "╚═══════════════════════════╝\n\n"
                        f"📦 Total: {result.get('success_count', 0)}\n"
                        f"🎉 These sessions migrated successfully"
                    )
                    await update.message.reply_document(
                        document=f,
                        filename='new_sessions_successful.zip',
                        caption=caption
                    )
                os.remove(success_zip)
            else:
                logger.warning(f"Skipping new ZIP - size too small ({size})")

        # Send old originals backup ZIP
        old_backup_zip = result.get('old_backup_zip')
        if old_backup_zip and os.path.exists(old_backup_zip):
            size = os.path.getsize(old_backup_zip)
            logger.info(f"Sending old backup ZIP: {old_backup_zip} (size: {size} bytes)")
            if size > 0:
                with open(old_backup_zip, 'rb') as f:
                    caption = (
                        "╔═══════════════════════════╗\n"
                        "║  📂 SUCCESSFUL OLD BACKUP  ║\n"
                        "╚═══════════════════════════╝\n\n"
                        f"📦 Total: {result.get('success_count', 0)}\n"
                        f"Original files of successful migrations (backup)"
                    )
                    await update.message.reply_document(
                        document=f,
                        filename='successful_old_original.zip',
                        caption=caption
                    )
                os.remove(old_backup_zip)
            else:
                logger.warning(f"Skipping old backup ZIP - size too small ({size})")

        # Send failed ZIP
        failed_zip = result.get('failed_zip')
        if failed_zip and os.path.exists(failed_zip):
            size = os.path.getsize(failed_zip)
            logger.info(f"Sending failed ZIP: {failed_zip} (size: {size} bytes)")
            if size > 0:
                with open(failed_zip, 'rb') as f:
                    caption = (
                        "╔═══════════════════════════╗\n"
                        "║  ❌ FAILED SESSIONS        ║\n"
                        "╚═══════════════════════════╝\n\n"
                        f"📦 Total: {result.get('failed_count', 0)}\n"
                        f"⚠️ These sessions could not be migrated"
                    )
                    await update.message.reply_document(
                        document=f,
                        filename='failed_old_sessions.zip',
                        caption=caption
                    )
                os.remove(failed_zip)
            else:
                logger.warning(f"Skipping failed ZIP - size too small ({size})")

        summary = (
            "╔═══════════════════════════╗\n"
            "║  📊 PROCESS COMPLETE       ║\n"
            "╚═══════════════════════════╝\n\n"
            f"📈 Total Sessions: {result.get('total', 0)}\n"
            f"✅ Successful: {result.get('success_count', 0)}\n"
            f"❌ Failed: {result.get('failed_count', 0)}\n\n"
            "🎯 Check ZIP files above for results!"
        )
        await update.message.reply_text(summary)

        cleanup(temp_zip_path, extract_dir)
        user_states[user_id] = None
        user_passwords[user_id] = None

    except Exception as e:
        logger.error(f"Critical error in handle_document for user {user_id}: {str(e)}", exc_info=True)
        await processing_msg.edit_text(
            "╔═══════════════════════════╗\n"
            "║  ❌ CRITICAL ERROR         ║\n"
            "╚═══════════════════════════╝\n\n"
            f"💥 Error: {str(e)}\n\n"
            "Please try again or contact support."
        )
        cleanup(temp_zip_path, extract_dir)
        user_states[user_id] = None
        user_passwords[user_id] = None

async def process_sessions(session_files, user_id, processing_msg, update, password):
    """
    Process multiple sessions with parallel processing.
    Returns 3 ZIP files even if some sessions hang or fail:
    - successful_new_sessions.zip     → newly migrated sessions (with country breakdown in caption)
    - successful_old_backup.zip       → backup of original successful sessions
    - failed_old_sessions.zip         → original files that failed
    """
    success_new_dir = os.path.join(tempfile.gettempdir(), f'success_new_{user_id}')
    success_old_dir = os.path.join(tempfile.gettempdir(), f'success_old_{user_id}')
    failed_old_dir  = os.path.join(tempfile.gettempdir(), f'failed_old_{user_id}')

    for d in [success_new_dir, success_old_dir, failed_old_dir]:
        os.makedirs(d, exist_ok=True)

    # Remove duplicates while preserving order
    seen = set()
    unique_files = []
    for f in session_files:
        if f not in seen:
            unique_files.append(f)
            seen.add(f)
    session_files = unique_files

    total = len(session_files)
    logger.info(f"Starting processing for {total} sessions | User: {user_id}")
    if total > 0:
        logger.info(f"Files to process: {[os.path.basename(f) for f in session_files]}")
    else:
        logger.warning("No session files to process")

    progress_state = {
        'total': total,
        'completed': 0,
        'success_count': 0,
        'failed_count': 0,
        'success_list': [],
        'success_paths': [],
        'successful_old_paths': [],
        'country_counts': {},  # New: Country code counts {'+880': 5, '+234': 2, ...}
        'lock': asyncio.Lock()
    }

    # Increased semaphore for faster processing (adjust based on testing)
    semaphore = asyncio.Semaphore(20)

    update_task = asyncio.create_task(update_progress_periodically(processing_msg, progress_state))

    async def process_single_session(session_file):
        async with semaphore:
            session_name = os.path.basename(session_file)
            logger.info(f"Processing: {session_name}")

            # Count completed at start
            async with progress_state['lock']:
                progress_state['completed'] += 1

            # Select random API from pool
            api = random.choice(API_POOL)

            try:
                result = await migrate_session(
                    session_file,
                    success_new_dir,
                    password,
                    user_id,
                    logout=False,
                    api_id=api["id"],
                    api_hash=api["hash"]
                )

                async with progress_state['lock']:
                    if result['success']:
                        progress_state['success_count'] += 1
                        progress_state['success_list'].append({
                            'phone': result['user_info']['phone'],
                            'name': result['user_info']['first_name']
                        })
                        new_path = result['new_session_path']
                        progress_state['success_paths'].append(new_path)
                        progress_state['successful_old_paths'].append(session_file)

                        shutil.copy(session_file, os.path.join(success_old_dir, session_name))
                        logger.info(f"Success: {session_name} → copied to backup")

                        country_code = result.get('country_code', '+1')
                        await insert_session_to_db(user_id, country_code, result['user_info']['phone'], new_path)

                        # Country count update
                        progress_state['country_counts'][country_code] = progress_state['country_counts'].get(country_code, 0) + 1

                    else:
                        progress_state['failed_count'] += 1
                        shutil.copy(session_file, os.path.join(failed_old_dir, session_name))
                        logger.warning(f"Failed: {session_name} - {result.get('error', 'No error')}")

            except asyncio.TimeoutError:
                async with progress_state['lock']:
                    progress_state['failed_count'] += 1
                    shutil.copy(session_file, os.path.join(failed_old_dir, session_name))
                    logger.error(f"Timeout: {session_name}")

            except Exception as e:
                async with progress_state['lock']:
                    progress_state['failed_count'] += 1
                    shutil.copy(session_file, os.path.join(failed_old_dir, session_name))
                    logger.error(f"Exception in {session_name}: {str(e)}")

    # Run parallel processing
    tasks = [process_single_session(sf) for sf in session_files]
    await asyncio.gather(*tasks, return_exceptions=True)

    # Cancel progress task
    update_task.cancel()
    try:
        await update_task
    except asyncio.CancelledError:
        pass

    # Final progress message
    async with progress_state['lock']:
        completed = min(progress_state['completed'], total)
        success = progress_state['success_count']
        failed = progress_state['failed_count']

    percentage = min(100, int((completed / total) * 100) if total > 0 else 0)
    progress_bar = create_progress_bar(percentage)
    processing_now = max(0, min(20, total - completed))

    await processing_msg.edit_text(
        "╔═══════════════════════════╗\n"
        "║  ✅ ALL SESSIONS PROCESSED ║\n"
        "╚═══════════════════════════╝\n\n"
        f"📊 Progress: {completed}/{total} ({percentage}%)\n"
        f"{progress_bar}\n\n"
        f"✅ Successful: {success}\n"
        f"❌ Failed: {failed}\n"
        f"⚡ Processing: {processing_now}\n\n"
        "Creating ZIP files...\n"
        "Performing batch logout on successful old sessions..."
    )

    # Batch logout successful old sessions
    if progress_state['successful_old_paths']:
        logger.info(f"Batch logout for {len(progress_state['successful_old_paths'])} sessions")
        await batch_logout_successful_old_sessions(progress_state['successful_old_paths'], user_id)

    # Create ZIPs
    new_zip    = os.path.join(tempfile.gettempdir(), f'new_success_{user_id}.zip')
    old_zip    = os.path.join(tempfile.gettempdir(), f'successful_old_backup_{user_id}.zip')
    failed_zip = os.path.join(tempfile.gettempdir(), f'failed_old_{user_id}.zip')

    # ZIP 1: New migrated sessions
    if progress_state['success_paths']:
        with zipfile.ZipFile(new_zip, 'w', zipfile.ZIP_DEFLATED) as zf:
            for path in progress_state['success_paths']:
                if os.path.exists(path):
                    zf.write(path, os.path.basename(path))

    # ZIP 2: Original successful backup
    with zipfile.ZipFile(old_zip, 'w', zipfile.ZIP_DEFLATED) as zf:
        for root, _, files in os.walk(success_old_dir):
            for file in files:
                full_path = os.path.join(root, file)
                arcname = os.path.relpath(full_path, success_old_dir)
                zf.write(full_path, arcname)

    # ZIP 3: Failed originals
    with zipfile.ZipFile(failed_zip, 'w', zipfile.ZIP_DEFLATED) as zf:
        for root, _, files in os.walk(failed_old_dir):
            for file in files:
                full_path = os.path.join(root, file)
                arcname = os.path.relpath(full_path, failed_old_dir)
                zf.write(full_path, arcname)

    # Send ZIPs safely
    for zip_path, fname, base_caption in [
        (new_zip,    'successful_new_sessions.zip',    f"New sessions ({success})"),
        (old_zip,    'successful_old_backup.zip',      f"Old originals backup ({success})"),
        (failed_zip, 'failed_old_sessions.zip',        f"Failed sessions ({failed})")
    ]:
        if zip_path and os.path.exists(zip_path) and os.path.getsize(zip_path) > 40:
            try:
                with open(zip_path, 'rb') as f:
                    # Special caption only for new sessions ZIP
                    if fname == 'successful_new_sessions.zip':
                        country_lines = []
                        for code, cnt in sorted(progress_state['country_counts'].items()):
                            country_lines.append(f"{code} - {cnt}")

                        caption = (
                            f"New sessions ({success})\n"
                            + "\n".join(country_lines) + "\n\n"
                            f"{datetime.now().strftime('%d/%m/%Y')}"
                        )
                    else:
                        caption = base_caption

                    await update.message.reply_document(
                        document=f,
                        filename=fname,
                        caption=caption
                    )
            except Exception as e:
                logger.error(f"Failed sending {fname}: {e}")
        else:
            logger.info(f"Skipping {fname}: empty or missing")

    # Cleanup
    for d in [success_new_dir, success_old_dir, failed_old_dir]:
        shutil.rmtree(d, ignore_errors=True)

    for z in [new_zip, old_zip, failed_zip]:
        if os.path.exists(z):
            os.remove(z)

    return {
        'total': total,
        'success_count': success,
        'failed_count': failed,
        'success_list': progress_state['success_list'],
        'success_zip': new_zip,
        'old_backup_zip': old_zip,
        'failed_zip': failed_zip,
        'country_counts': progress_state['country_counts']
    }

async def batch_logout_successful_old_sessions(successful_old_paths, user_id):
    """Batch logout all successful old sessions at the end"""
    if not successful_old_paths:
        logger.info("No successful sessions to logout.")
        return

    logger.info(f"Batch logout started for {len(successful_old_paths)} sessions | User: {user_id}")

    semaphore = asyncio.Semaphore(15)  # Slightly lower to reduce flood risk

    async def logout_single_old_session(old_path):
        async with semaphore:
            session_name = os.path.basename(old_path)
            # Select a random API from pool for logout
            api = random.choice(API_POOL)
            try:
                session_file = old_path.replace('.session', '')
                proxy = None  # You can add proxy selection logic here if needed

                old_client = TelegramClient(
                    session_file,
                    api["id"],
                    api["hash"],
                    proxy=proxy,
                    timeout=180,
                    connection_retries=8,
                    retry_delay=10
                )

                await old_client.connect()
                if await old_client.is_user_authorized():
                    await old_client.log_out()
                    logger.info(f"Logout success: {session_name}")
                else:
                    logger.info(f"Skipped (not authorized): {session_name}")

                await old_client.disconnect()

            except FloodWaitError as fwe:
                logger.warning(f"FloodWait {fwe.seconds}s for {session_name}")
                await asyncio.sleep(fwe.seconds + random.uniform(10, 30))
                # Optional: retry once
                try:
                    await old_client.connect()
                    await old_client.log_out()
                    logger.info(f"Retry success: {session_name}")
                except:
                    logger.error(f"Retry failed: {session_name}")

            except Exception as e:
                logger.error(f"Batch logout error {session_name}: {str(e)}")

    tasks = [logout_single_old_session(p) for p in successful_old_paths]
    await asyncio.gather(*tasks, return_exceptions=True)

    logger.info("Batch logout completed.")

async def update_progress_periodically(processing_msg, progress_state):
    """Update progress message periodically"""
    last_completed = -1

    while True:
        try:
            await asyncio.sleep(2)

            async with progress_state['lock']:
                completed = progress_state['completed']
                total = progress_state['total']
                success = progress_state['success_count']
                failed = progress_state['failed_count']

            # Only update if progress changed
            if completed != last_completed:
                last_completed = completed

                percentage = int((completed / total) * 100) if total > 0 else 0
                progress_bar = create_progress_bar(percentage)

                await processing_msg.edit_text(
                    "╔═══════════════════════════╗\n"
                    "║  🚀 PARALLEL PROCESSING    ║\n"
                    "╚═══════════════════════════╝\n\n"
                    f"📊 Progress: {completed}/{total} ({percentage}%)\n"
                    f"{progress_bar}\n\n"
                    f"✅ Successful: {success}\n"
                    f"❌ Failed: {failed}\n"
                    f"⚡ Processing: {min(20, total - completed)}\n\n"
                    f"⏳ Please wait..."
                )

        except asyncio.CancelledError:
            break
        except Exception as e:
            print(f'Progress update error: {str(e)}')
            await asyncio.sleep(2)

def create_progress_bar(percentage, length=20):
    """Create a text-based progress bar"""
    filled = int(length * percentage / 100)
    bar = '█' * filled + '░' * (length - filled)
    return f"[{bar}]"

async def migrate_session(old_session_path, output_dir, password=None, user_id=None, logout=True, api_id=None, api_hash=None):
    """
    Migrate old session to new session using event-driven OTP listener.
    IMPORTANT:
    - Old .session file is NOT deleted (needed for successful_old_original.zip)
    - New session filename is unique (no overwrite)
    """
    old_client = None
    new_client = None
    country_code = '+1'
    otp_queue = asyncio.Queue()  # Queue to hold OTP from event

    session_name = os.path.basename(old_session_path)
    logger.info(f"Starting migration: {session_name}")

    try:
        session_file = old_session_path.replace('.session', '')

        # Step 1: Read phone from old session (no proxy, using provided API)
        temp_client = TelegramClient(session_file, api_id or config.API_ID, api_hash or config.API_HASH)
        await asyncio.wait_for(temp_client.connect(), timeout=60)

        if not await temp_client.is_user_authorized():
            raise Exception("Old session not authorized or expired")

        me = await asyncio.wait_for(temp_client.get_me(), timeout=30)
        await temp_client.disconnect()

        phone = me.phone or str(me.id)
        if not phone.startswith('+'):
            phone = '+' + phone

        # Parse country code
        try:
            parsed = phonenumbers.parse(phone)
            country_code = f"+{parsed.country_code}"
        except Exception:
            country_code = '+1'

        # Select proxy by country
        proxy = user_proxies.get(user_id, {}).get(country_code)

        logger.info(f"Connected as {me.first_name} ({phone})")

        # Step 2: Connect old client (with proxy if exists)
        old_client = TelegramClient(session_file, api_id or config.API_ID, api_hash or config.API_HASH, proxy=proxy)
        await asyncio.wait_for(old_client.connect(), timeout=config.TELETHON_TIMEOUT)

        if not await old_client.is_user_authorized():
            raise Exception("Old session authorization lost")

        # Step 3: Setup event-driven OTP listener on old_client for Telegram official (777000)
        @old_client.on(events.NewMessage(from_users=777000))
        async def otp_handler(event):
            message_text = event.message.message
            codes = re.findall(r'\b\d{5,6}\b', message_text)
            if codes:
                otp = codes[0]
                logger.info(f"Received OTP via event: {otp} for {session_name}")
                await otp_queue.put(otp)  # Put OTP in queue

        # Start the client to listen for events
        await old_client.start()

        # Step 4: Create NEW session (unique name, no overwrite)
        new_session_path = os.path.join(
            output_dir,
            f"{phone}.session"
        )

        new_client = TelegramClient(
            new_session_path.replace('.session', ''),
            api_id or config.API_ID,
            api_hash or config.API_HASH,
            proxy=proxy
        )
        await asyncio.wait_for(new_client.connect(), timeout=config.TELETHON_TIMEOUT)

        # Step 5: Send OTP request
        sent_code = await asyncio.wait_for(
            new_client.send_code_request(phone),
            timeout=60
        )
        phone_code_hash = sent_code.phone_code_hash

        # Step 6: Wait for OTP from event queue with timeout
        try:
            otp_code = await asyncio.wait_for(otp_queue.get(), timeout=config.OTP_MAX_WAIT_SECONDS)
        except asyncio.TimeoutError:
            raise Exception("OTP not received within timeout")

        # Step 7: Sign in
        try:
            await asyncio.wait_for(
                new_client.sign_in(phone, otp_code, phone_code_hash=phone_code_hash),
                timeout=60
            )
        except SessionPasswordNeededError:
            if not password:
                raise Exception("2FA password required but not provided")
            await new_client.sign_in(password=password)

        # Step 8: Verify new session
        new_me = await asyncio.wait_for(new_client.get_me(), timeout=30)
        if not new_me:
            raise Exception("New session verification failed")

        logger.info(f"New session verified: {new_me.first_name} ({new_me.phone})")

        # Step 9: Logout old session (file is kept!)
        if logout:
            try:
                await asyncio.wait_for(old_client.log_out(), timeout=30)
                logger.info(f"Old session logged out: {session_name}")
            except Exception as e:
                logger.warning(f"Logout failed (ignored): {e}")

        # Remove event handler to clean up
        old_client.remove_event_handler(otp_handler, events.NewMessage)

        await old_client.disconnect()
        await new_client.disconnect()

        return {
            'success': True,
            'new_session_path': new_session_path,
            'user_info': {
                'id': str(new_me.id),
                'first_name': new_me.first_name or '',
                'last_name': new_me.last_name or '',
                'username': new_me.username or '',
                'phone': phone,
            },
            'country_code': country_code
        }

    except Exception as e:
        logger.error(f"Migration failed {session_name}: {e}")

        if old_client and old_client.is_connected():
            old_client.remove_event_handler(otp_handler, events.NewMessage) if 'otp_handler' in locals() else None
            await old_client.disconnect()
        if new_client and new_client.is_connected():
            await new_client.disconnect()

        return {
            'success': False,
            'error': str(e)
        }

    """Handle /logout command"""

async def handle_logout_zip(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle uploaded ZIP for logout"""
    user_id = update.effective_user.id

    if user_states.get(user_id) != 'waiting_logout_zip':
        return  # Not in logout mode, ignore

    document = update.message.document
    if not document or not document.file_name.lower().endswith('.zip'):
        await update.message.reply_text("Please send a valid .zip file containing .session files.")
        return

    user_states[user_id] = 'processing_logout'

    processing_msg = await update.message.reply_text("⏳ Processing logout ZIP...")

    try:
        file = await context.bot.get_file(document.file_id)
        temp_zip = os.path.join(tempfile.gettempdir(), f'logout_{user_id}.zip')
        await file.download_to_drive(temp_zip)

        extract_dir = os.path.join(tempfile.gettempdir(), f'logout_extract_{user_id}')
        os.makedirs(extract_dir, exist_ok=True)

        with zipfile.ZipFile(temp_zip, 'r') as z:
            z.extractall(extract_dir)

        session_files = []
        for root, _, files in os.walk(extract_dir):
            for f in files:
                if f.endswith('.session'):
                    session_files.append(os.path.join(root, f))

        if not session_files:
            await processing_msg.edit_text("No .session files found in ZIP!")
            cleanup(temp_zip, extract_dir)
            user_states[user_id] = None
            return

        total = len(session_files)
        await processing_msg.edit_text(f"Found {total} sessions. Logging out...")

        success_dir = os.path.join(tempfile.gettempdir(), f'logout_success_{user_id}')
        failed_dir = os.path.join(tempfile.gettempdir(), f'logout_failed_{user_id}')
        os.makedirs(success_dir, exist_ok=True)
        os.makedirs(failed_dir, exist_ok=True)

        success_count = failed_count = 0
        sem = asyncio.Semaphore(10)

        async def logout_one(path):
            nonlocal success_count, failed_count
            async with sem:
                # Select a random API from pool for logout
                api = random.choice(API_POOL)
                try:
                    name = os.path.basename(path)
                    client = TelegramClient(name.replace('.session', ''), api["id"], api["hash"])
                    await client.connect()
                    if await client.is_user_authorized():
                        await client.log_out()
                        success_count += 1
                        shutil.copy(path, os.path.join(success_dir, name))
                    else:
                        failed_count += 1
                        shutil.copy(path, os.path.join(failed_dir, name))
                    await client.disconnect()
                except Exception as e:
                    failed_count += 1
                    shutil.copy(path, os.path.join(failed_dir, name))
                    print(f"Logout error {name}: {e}")

        await asyncio.gather(*[logout_one(f) for f in session_files])

        # ZIPs
        success_zip = create_zip_from_dir(success_dir, f'logout_success_{user_id}.zip')
        failed_zip = create_zip_from_dir(failed_dir, f'logout_failed_{user_id}.zip')

        if success_zip:
            await update.message.reply_document(open(success_zip, 'rb'), filename='logout_success.zip', caption=f"Success: {success_count}")
        if failed_zip:
            await update.message.reply_document(open(failed_zip, 'rb'), filename='logout_failed.zip', caption=f"Failed: {failed_count}")

        await processing_msg.edit_text(f"Logout complete! Success: {success_count} | Failed: {failed_count}")

        cleanup(temp_zip, extract_dir, success_dir, failed_dir, success_zip, failed_zip)
        user_states[user_id] = None

    except Exception as e:
        await processing_msg.edit_text(f"Error: {str(e)}")
        user_states[user_id] = None

def cleanup(*paths):
    """Safe cleanup for files and directories"""
    for path in paths:
        try:
            if os.path.isfile(path):
                os.remove(path)
            elif os.path.isdir(path):
                shutil.rmtree(path, ignore_errors=True)
        except Exception as e:
            logger.error(f"Cleanup error for {path}: {e}")

def create_zip_from_dir(source_dir, zip_name):
    """Create ZIP from directory"""
    zip_path = os.path.join(tempfile.gettempdir(), zip_name)
    try:
        with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zf:
            for root, _, files in os.walk(source_dir):
                for file in files:
                    full = os.path.join(root, file)
                    arc = os.path.relpath(full, source_dir)
                    zf.write(full, arc)
        return zip_path
    except Exception as e:
        logger.error(f"ZIP creation failed: {e}")
        return None
